cmake_minimum_required(VERSION 3.24)
project(stx_skia VERSION 0.1.0 LANGUAGES C CXX)
include(CTest)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
include(cmake/sources.cmake)
option(STX_SKIA_BASELINE "Build the unpruned source closure for comparison" OFF)
option(STX_SKIA_SANITIZE "Instrument the complete library and tests" OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(stx_sources ${STX_SKIA_BASELINE_SOURCES})
if(NOT APPLE)
    list(FILTER stx_sources EXCLUDE REGEX "skia/src/(ports/SkImageGeneratorCG|utils/mac/).*\\.cpp$")
endif()
if(WIN32)
    list(REMOVE_ITEM stx_sources skia/src/ports/SkOSFile_posix.cpp)
    list(APPEND stx_sources skia/src/ports/SkOSFile_win.cpp skia/src/ports/SkLog_win.cpp)
endif()
if(STX_SKIA_BASELINE AND CMAKE_SYSTEM_PROCESSOR MATCHES "x86_64|AMD64|amd64")
    list(APPEND stx_sources skia/src/opts/SkOpts_ml3.cpp)
endif()
if(STX_SKIA_BASELINE)
    list(REMOVE_ITEM stx_sources skia/src/core/SkData.cpp)
    list(APPEND stx_sources baseline/SkData.cpp)
    include(cmake/reference.cmake)
else()
    include(cmake/prune.cmake)
endif()
add_library(stx_skia STATIC ${stx_sources} src/pixel_storage.cpp)
add_library(stx_skia::stx_skia ALIAS stx_skia)
target_compile_features(stx_skia PUBLIC cxx_std_20)
if(STX_SKIA_BASELINE)
    target_include_directories(stx_skia BEFORE PUBLIC
        $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/baseline/skia>)
endif()
target_include_directories(stx_skia PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/skia>
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}/stx_skia/skia>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>)
target_compile_definitions(stx_skia PUBLIC SK_DISABLE_TRACING SK_GAMMA_APPLY_TO_A8)
target_compile_definitions(stx_skia PRIVATE SKIA_IMPLEMENTATION=1 SK_ENABLE_PRECOMPILE
    SKCMS_DISABLE_HSW SKCMS_DISABLE_SKX)
if(NOT STX_SKIA_BASELINE)
    target_compile_definitions(stx_skia PUBLIC SK_DISABLE_EFFECT_DESERIALIZATION)
    target_compile_definitions(stx_skia PRIVATE STX_SKIA_BASELINE_SIMD)
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
    target_compile_options(stx_skia PRIVATE -ffp-contract=off -fno-exceptions -fno-rtti
        -fvisibility=hidden -fvisibility-inlines-hidden -Wno-attributes)
    if(WIN32 AND CMAKE_CXX_COMPILER_ID MATCHES "Clang")
        target_compile_options(stx_skia PUBLIC -fms-extensions)
    endif()
endif()
set_target_properties(stx_skia PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(APPLE)
    target_link_libraries(stx_skia PUBLIC "-framework CoreFoundation" "-framework CoreGraphics")
    if(STX_SKIA_BASELINE)
        target_link_libraries(stx_skia PUBLIC "-framework ApplicationServices")
    endif()
endif()
if(STX_SKIA_SANITIZE)
    # Skia supplies no RTTI, so vptr instrumentation cannot cross this boundary.
    # Do not disable consumer RTTI: GUI.Forms uses it for its own adapter types.
    target_compile_options(stx_skia PUBLIC -fsanitize=address,undefined
        -fno-sanitize=vptr -fno-omit-frame-pointer)
    target_link_options(stx_skia PUBLIC -fsanitize=address,undefined)
endif()
if(BUILD_TESTING)
    find_package(Threads REQUIRED)
    add_executable(stx_skia_render_tests tests/render.cpp)
    target_link_libraries(stx_skia_render_tests PRIVATE stx_skia)
    add_test(NAME render_contract COMMAND stx_skia_render_tests)
    add_executable(stx_skia_storage_tests tests/pixel_storage.cpp)
    target_link_libraries(stx_skia_storage_tests PRIVATE stx_skia)
    add_test(NAME pixel_storage COMMAND stx_skia_storage_tests)
    add_executable(stx_skia_data_tests tests/data.cpp)
    target_link_libraries(stx_skia_data_tests PRIVATE stx_skia)
    add_test(NAME data_contract COMMAND stx_skia_data_tests)
    add_executable(stx_skia_pixel_ref_tests tests/pixel_ref.cpp)
    target_link_libraries(stx_skia_pixel_ref_tests PRIVATE stx_skia Threads::Threads)
    add_test(NAME pixel_ref_contract COMMAND stx_skia_pixel_ref_tests)
    add_executable(stx_skia_ownership_view_tests tests/ownership_views.cpp)
    target_link_libraries(stx_skia_ownership_view_tests PRIVATE stx_skia Threads::Threads)
    add_test(NAME ownership_view_contract COMMAND stx_skia_ownership_view_tests)
    add_executable(stx_skia_pixel_extent_tests tests/pixel_extent.cpp)
    target_link_libraries(stx_skia_pixel_extent_tests PRIVATE stx_skia)
    add_test(NAME pixel_extent_contract COMMAND stx_skia_pixel_extent_tests)
    add_executable(stx_skia_pixel_conversion_tests tests/pixel_conversion.cpp)
    target_link_libraries(stx_skia_pixel_conversion_tests PRIVATE stx_skia)
    add_test(NAME pixel_conversion_contract COMMAND stx_skia_pixel_conversion_tests)
    add_executable(stx_skia_pixmap_tests tests/pixmap.cpp)
    target_link_libraries(stx_skia_pixmap_tests PRIVATE stx_skia)
    add_test(NAME pixmap_contract COMMAND stx_skia_pixmap_tests)
    add_executable(stx_skia_mask_tests tests/mask.cpp)
    target_link_libraries(stx_skia_mask_tests PRIVATE stx_skia)
    add_test(NAME mask_contract COMMAND stx_skia_mask_tests)
    add_executable(stx_skia_cached_data_tests tests/cached_data.cpp)
    target_link_libraries(stx_skia_cached_data_tests PRIVATE stx_skia Threads::Threads)
    add_test(NAME cached_data_contract COMMAND stx_skia_cached_data_tests)
    add_executable(stx_skia_mask_filter_tests tests/mask_filter.cpp)
    target_link_libraries(stx_skia_mask_filter_tests PRIVATE stx_skia)
    add_test(NAME mask_filter_contract COMMAND stx_skia_mask_filter_tests)
    if(STX_SKIA_BASELINE)
        target_compile_definitions(stx_skia_mask_filter_tests PRIVATE STX_SKIA_REFERENCE=1)
        target_compile_definitions(stx_skia_cached_data_tests PRIVATE STX_SKIA_REFERENCE=1)
        target_compile_definitions(stx_skia_mask_tests PRIVATE STX_SKIA_REFERENCE=1)
        target_compile_definitions(stx_skia_pixmap_tests PRIVATE STX_SKIA_REFERENCE=1)
        target_compile_definitions(stx_skia_pixel_conversion_tests PRIVATE STX_SKIA_REFERENCE=1)
        target_compile_definitions(stx_skia_pixel_extent_tests PRIVATE STX_SKIA_REFERENCE=1)
        target_compile_definitions(stx_skia_ownership_view_tests PRIVATE STX_SKIA_REFERENCE=1)
    endif()
    if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
        target_compile_options(stx_skia_mask_filter_tests PRIVATE -fno-rtti)
        target_compile_options(stx_skia_cached_data_tests PRIVATE -fno-rtti)
        target_compile_options(stx_skia_pixel_ref_tests PRIVATE -fno-rtti)
        target_compile_options(stx_skia_ownership_view_tests PRIVATE -fno-rtti)
    endif()
endif()

install(TARGETS stx_skia EXPORT stx_skiaTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(DIRECTORY skia/include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/stx_skia/skia/include
    FILES_MATCHING PATTERN "*.h" PATTERN "gpu" EXCLUDE PATTERN "third_party" EXCLUDE)
install(FILES skia/modules/skcms/skcms.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/stx_skia/skia/modules/skcms)
install(DIRECTORY skia/modules/skcms/src/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR}/stx_skia/skia/modules/skcms/src FILES_MATCHING PATTERN "*.h")
install(FILES LICENSE LICENSE-Skia DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/stx_skia)
install(EXPORT stx_skiaTargets NAMESPACE stx_skia:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/stx_skia)
configure_package_config_file(cmake/stx_skiaConfig.cmake.in
    "${CMAKE_CURRENT_BINARY_DIR}/stx_skiaConfig.cmake"
    INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/stx_skia)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/stx_skiaConfigVersion.cmake"
    VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/stx_skiaConfig.cmake"
    "${CMAKE_CURRENT_BINARY_DIR}/stx_skiaConfigVersion.cmake" DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/stx_skia)
